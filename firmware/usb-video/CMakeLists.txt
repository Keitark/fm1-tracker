cmake_minimum_required(VERSION 3.20)
project(fm1_uvc_stream C)
enable_testing()
add_library(fm1_uvc_stream uvc_stream.c uvc_control.c)
target_include_directories(fm1_uvc_stream PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(fm1_uvc_stream PUBLIC c_std_11)
add_executable(uvc_stream_tests tests/test_uvc_stream.c)
target_link_libraries(uvc_stream_tests PRIVATE fm1_uvc_stream)
add_executable(uvc_control_tests tests/test_uvc_control.c)
target_link_libraries(uvc_control_tests PRIVATE fm1_uvc_stream)
add_library(fm1_uvc_stream_ipad_pad uvc_stream.c uvc_control.c)
target_include_directories(fm1_uvc_stream_ipad_pad PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(fm1_uvc_stream_ipad_pad PUBLIC c_std_11)
target_compile_definitions(fm1_uvc_stream_ipad_pad PUBLIC FM1_UVC_IPAD_PAD=1)
add_executable(uvc_stream_ipad_pad_tests tests/test_uvc_stream.c)
target_link_libraries(uvc_stream_ipad_pad_tests PRIVATE fm1_uvc_stream_ipad_pad)
add_executable(uvc_control_ipad_pad_tests tests/test_uvc_control.c)
target_link_libraries(uvc_control_ipad_pad_tests PRIVATE fm1_uvc_stream_ipad_pad)
foreach(target fm1_uvc_stream uvc_stream_tests uvc_control_tests
               fm1_uvc_stream_ipad_pad uvc_stream_ipad_pad_tests uvc_control_ipad_pad_tests)
if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endif()
endforeach()
add_test(NAME uvc_stream_bounded_queue_and_yuy2 COMMAND uvc_stream_tests)
add_test(NAME uvc_descriptors_probe_commit_alt_and_reset COMMAND uvc_control_tests)
add_test(NAME uvc_ipad_pad_stream_raster_and_lifecycle COMMAND uvc_stream_ipad_pad_tests)
add_test(NAME uvc_ipad_pad_descriptors_and_control COMMAND uvc_control_ipad_pad_tests)
