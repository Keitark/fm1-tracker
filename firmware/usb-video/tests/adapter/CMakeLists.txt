cmake_minimum_required(VERSION 3.20)
project(fm1_uvc_adapter_tests C)
enable_testing()
add_executable(uvc_adapter_tests ../test_uvc_adapter.c ../../fm1_uvc.c ../../uvc_control.c ../../uvc_stream.c)
target_include_directories(uvc_adapter_tests PRIVATE ../..)
target_compile_definitions(uvc_adapter_tests PRIVATE FM1_UVC_SCREEN=1 FM1_UVC_ADAPTER_TEST=1)
target_compile_features(uvc_adapter_tests PRIVATE c_std_11)
add_executable(uvc_adapter_ipad_pad_tests ../test_uvc_adapter.c ../../fm1_uvc.c ../../uvc_control.c ../../uvc_stream.c)
target_include_directories(uvc_adapter_ipad_pad_tests PRIVATE ../..)
target_compile_definitions(uvc_adapter_ipad_pad_tests PRIVATE FM1_UVC_SCREEN=1 FM1_UVC_ADAPTER_TEST=1 FM1_UVC_IPAD_PAD=1)
target_compile_features(uvc_adapter_ipad_pad_tests PRIVATE c_std_11)
foreach(target uvc_adapter_tests uvc_adapter_ipad_pad_tests)
if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endif()
endforeach()
add_test(NAME sdk_uvc_adapter_dma_and_recovery_lifecycle COMMAND uvc_adapter_tests)
add_test(NAME sdk_uvc_ipad_pad_adapter_dma_and_recovery COMMAND uvc_adapter_ipad_pad_tests)
foreach(profile IN ITEMS cdc uac)
    add_executable(uvc_forge_policy_${profile} ../test_uvc_adapter.c ../../fm1_uvc.c ../../uvc_control.c ../../uvc_stream.c ../../../usb-diag/usb_policy.c)
    target_include_directories(uvc_forge_policy_${profile} PRIVATE ../..)
    target_compile_definitions(uvc_forge_policy_${profile} PRIVATE FM1_USB_VIDEO=1 FM1_UVC_ADAPTER_TEST=1 FM1_UVC_IPAD_PAD=1 FM1_USB_POLICY_HOST=1 FM1_UVC_POLICY_TEST=1)
    if(profile STREQUAL uac)
        target_compile_definitions(uvc_forge_policy_${profile} PRIVATE FM1_USB_AUDIO=1)
    endif()
    target_compile_features(uvc_forge_policy_${profile} PRIVATE c_std_11)
    if(MSVC)
        target_compile_options(uvc_forge_policy_${profile} PRIVATE /W4 /WX)
    else()
        target_compile_options(uvc_forge_policy_${profile} PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME sdk_uvc_forge_policy_${profile} COMMAND uvc_forge_policy_${profile})
endforeach()
