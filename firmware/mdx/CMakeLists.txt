cmake_minimum_required(VERSION 3.20)
project(fm1_mdx C)
enable_testing()
file(GLOB VENDOR_SOURCES CONFIGURE_DEPENDS vendor/retrofm/*.c vendor/mdxtools/*.c)
add_library(fm1_mdx src/fm1_mdx.c src/fm1_mdx_usb.c ${VENDOR_SOURCES} samples/demo.c)
target_include_directories(fm1_mdx PUBLIC include vendor/retrofm vendor/mdxtools)
target_compile_features(fm1_mdx PUBLIC c_std_11)
if(NOT MSVC)
  target_link_libraries(fm1_mdx PUBLIC m)
endif()
add_executable(mdx_tests tests/test_mdx.c)
target_link_libraries(mdx_tests PRIVATE fm1_mdx)
add_test(NAME mdx_karaoke_usb COMMAND mdx_tests)
add_executable(mdx_usb_pipe tests/usb_pipe.c ../usb-diag/protocol.c)
target_include_directories(mdx_usb_pipe PRIVATE ../usb-diag)
target_compile_definitions(mdx_usb_pipe PRIVATE FM1_MDX_PLAYER=1 FM1_PERIPHERAL_TESTS=1)
target_link_libraries(mdx_usb_pipe PRIVATE fm1_mdx)
add_executable(mdx_target_tests tests/target_runtime.c ../usb-diag/peripheral_logic.c ../nes/src/fm1_stock_keys.c ../nes/src/fm1_volume.c ../nes/src/fm1_audio_queue.c)
target_include_directories(mdx_target_tests PRIVATE ../usb-diag ../usb-diag/tests ../nes/include ../nes/boot)
target_compile_definitions(mdx_target_tests PRIVATE FM1_MDX_PLAYER=1 FM1_KEYSCAN_IRQ=1 FM1_KEYSCAN_PACED=1 FM1_KEYSCAN_DMA2=1 FM1_VOLUME_TEST=1)
target_link_libraries(mdx_target_tests PRIVATE fm1_mdx)
add_test(NAME mdx_target_lifecycle COMMAND mdx_target_tests)
add_executable(mdx_render tests/render.c)
target_link_libraries(mdx_render PRIVATE fm1_mdx)
