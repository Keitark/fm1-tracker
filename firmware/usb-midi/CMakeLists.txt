cmake_minimum_required(VERSION 3.16)
project(fm1_usb_midi C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(midi_codec_tests tests/codec.c fm1_midi.c)
target_include_directories(midi_codec_tests PRIVATE .)
add_test(NAME common_midi_codec_descriptors COMMAND midi_codec_tests)
foreach(mode IN ITEMS blocked qualified)
  add_executable(midi_update_${mode}_tests tests/update.c fm1_midi_update.c fm1_sha256.c)
  target_include_directories(midi_update_${mode}_tests PRIVATE .)
  target_compile_definitions(midi_update_${mode}_tests PRIVATE FM1_MIDI_UPDATE_HOST=1)
  if(mode STREQUAL "qualified")
    target_compile_definitions(midi_update_${mode}_tests PRIVATE FM1_MIDI_LOADER_COMPATIBLE=1)
  endif()
  add_test(NAME common_midi_sdk_${mode} COMMAND midi_update_${mode}_tests)
endforeach()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../usb-diag/app_config.h")
  add_executable(midi_target_tests tests/target.c fm1_midi_target.c fm1_midi.c fm1_midi_update.c fm1_sha256.c)
  target_include_directories(midi_target_tests PRIVATE .)
  target_compile_definitions(midi_target_tests PRIVATE FM1_MIDI_TARGET_HOST=1 FM1_MIDI_UPDATE_HOST=1)
  add_test(NAME common_midi_actual_usb_target COMMAND midi_target_tests)
else()
  message(STATUS "Skipping parent USB target integration test: ../usb-diag/app_config.h is not included")
endif()
add_executable(midi_sha256_tests tests/sha256.c fm1_sha256.c)
target_include_directories(midi_sha256_tests PRIVATE .)
add_test(NAME common_midi_sha256 COMMAND midi_sha256_tests)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../usb-diag/usb_policy.c")
  foreach(profile IN ITEMS cdc uac)
    add_executable(midi_policy_${profile}_tests tests/policy.c)
    target_compile_definitions(midi_policy_${profile}_tests PRIVATE FM1_USB_MIDI_UPDATE=1)
    if(profile STREQUAL "uac")
      target_compile_definitions(midi_policy_${profile}_tests PRIVATE FM1_USB_AUDIO=1)
    endif()
    add_test(NAME common_midi_composite_policy_${profile} COMMAND midi_policy_${profile}_tests)
  endforeach()
else()
  message(STATUS "Skipping parent composite policy integration tests: ../usb-diag/usb_policy.c is not included")
endif()
if(MSVC)
  foreach(t IN ITEMS midi_codec_tests midi_update_blocked_tests midi_update_qualified_tests midi_target_tests)
    if(TARGET ${t})
      target_compile_definitions(${t} PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
  endforeach()
endif()
