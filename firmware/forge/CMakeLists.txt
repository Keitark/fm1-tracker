cmake_minimum_required(VERSION 3.20)
project(fm1_forge_host LANGUAGES C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(forge_preview tests/preview.c render.c forge.c)
target_include_directories(forge_preview PRIVATE .)
if(MSVC)
    target_compile_definitions(forge_preview PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_executable(forge_behavior tests/behavior.c forge.c render.c)
target_include_directories(forge_behavior PRIVATE .)
if(MSVC)
    target_compile_definitions(forge_behavior PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(forge_behavior PRIVATE /W4)
endif()
add_test(NAME forge_inputs_tone_and_renderer COMMAND forge_behavior)
add_executable(forge_benchmark tests/benchmark.c benchmark.c)
target_include_directories(forge_benchmark PRIVATE .)
if(MSVC)
    target_compile_definitions(forge_benchmark PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(forge_benchmark PRIVATE /W4)
endif()
add_test(NAME forge_deterministic_foreground_benchmark COMMAND forge_benchmark)
foreach(profile IN ITEMS default clock320 user cdc user_no_buttons user_no_encoders user_no_volume user_no_lcd user_no_audio user_no_hal diagnostics_no_hal)
    if(profile MATCHES "^user")
        set(module_source tests/user_module.c)
    else()
        set(module_source user_module.c)
    endif()
    add_executable(forge_runtime_${profile} tests/runtime.c forge.c render.c benchmark.c ${module_source}
        ../usb-diag/peripheral_logic.c ../nes/src/fm1_stock_keys.c ../nes/src/fm1_audio_queue.c)
    target_include_directories(forge_runtime_${profile} PRIVATE . ../usb-diag ../usb-audio ../nes/include ../nes/boot)
    target_compile_definitions(forge_runtime_${profile} PRIVATE FM1_FORGE_HOST=1 FM1_FORGE=1 FM1_KEYSCAN_IRQ=1 FM1_KEYSCAN_DMA2=1 FM1_KEYSCAN_PACED=1 FM1_LCD_ASYNC=1)
    if(NOT profile STREQUAL cdc)
        target_compile_definitions(forge_runtime_${profile} PRIVATE FM1_USB_AUDIO=1)
    endif()
    if(profile MATCHES "^user")
        target_compile_definitions(forge_runtime_${profile} PRIVATE FM1_FORGE_USER_MODULE=1)
    endif()
    foreach(hal IN ITEMS buttons encoders volume lcd audio)
        if(profile STREQUAL "user_no_${hal}" OR profile MATCHES "_no_hal$")
            string(TOUPPER ${hal} hal_upper)
            target_compile_definitions(forge_runtime_${profile} PRIVATE FM1_FORGE_HAL_${hal_upper}=0)
        endif()
    endforeach()
    if(profile STREQUAL clock320)
        target_compile_definitions(forge_runtime_${profile} PRIVATE FM1_FORGE_BOOT_CLOCK_HZ=320000000)
    endif()
    if(MSVC)
        target_compile_definitions(forge_runtime_${profile} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(forge_runtime_${profile} PRIVATE /W4)
    endif()
    add_test(NAME forge_actual_runtime_${profile} COMMAND forge_runtime_${profile})
endforeach()
foreach(lcd IN ITEMS stock rgb444 spi15 spi30)
    add_executable(forge_lcd_wire_${lcd} ../nes/tests/display_tests.c ../nes/boot/display_test.c render.c forge.c)
    target_include_directories(forge_lcd_wire_${lcd} PRIVATE . ../nes/boot)
    target_compile_definitions(forge_lcd_wire_${lcd} PRIVATE FM1_FORGE_WIRE_TEST=1 FM1_FORGE=1
        FM1_DISPLAY_TEST_HOST=1 FM1_LCD_STOCK_FILL=1 FM1_LCD_STOCK_DMA=1 FM1_LCD_STOCK_SEQUENCE=1
        FM1_NES_PLAYER=1 FM1_LCD_ASYNC=1)
    if(NOT lcd STREQUAL stock)
        target_compile_definitions(forge_lcd_wire_${lcd} PRIVATE FM1_LCD_RGB444=1)
    endif()
    if(lcd STREQUAL spi15)
        target_compile_definitions(forge_lcd_wire_${lcd} PRIVATE FM1_LCD_SPI15=1)
    elseif(lcd STREQUAL spi30)
        target_compile_definitions(forge_lcd_wire_${lcd} PRIVATE FM1_LCD_SPI30=1)
    endif()
    if(MSVC)
        target_compile_definitions(forge_lcd_wire_${lcd} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(forge_lcd_wire_${lcd} PRIVATE /W4)
    endif()
    add_test(NAME forge_actual_lcd_wire_${lcd} COMMAND forge_lcd_wire_${lcd})
endforeach()
