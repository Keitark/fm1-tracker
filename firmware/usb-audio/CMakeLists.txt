cmake_minimum_required(VERSION 3.20)
project(fm1_uac_capture_tests C)
enable_testing()
foreach(profile IN ITEMS cdc cdc_uvc)
    add_executable(uac_capture_${profile} tests/test_uac_capture.c fm1_usb_audio_capture.c fm1_usb_audio_profile.c)
    target_include_directories(uac_capture_${profile} PRIVATE .)
    target_compile_definitions(uac_capture_${profile} PRIVATE FM1_USB_AUDIO_CAPTURE_TEST=1)
    if(profile STREQUAL "cdc_uvc")
        target_compile_definitions(uac_capture_${profile} PRIVATE FM1_UVC_SCREEN=1)
    endif()
    target_compile_features(uac_capture_${profile} PRIVATE c_std_11)
    if(MSVC)
        target_compile_options(uac_capture_${profile} PRIVATE /W4 /WX)
    else()
        target_compile_options(uac_capture_${profile} PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME capture_${profile}_profile_conversion_fifo_epochs_clock COMMAND uac_capture_${profile})
endforeach()
