cmake_minimum_required(VERSION 3.20)
project(fm1_uac_adapter_tests C)
enable_testing()
foreach(profile IN ITEMS cdc cdc_uvc)
    add_executable(uac_adapter_${profile} ../test_uac_adapter.c ../../fm1_usb_audio_target.c ../../fm1_usb_audio_capture.c ../../fm1_usb_audio_profile.c)
    target_include_directories(uac_adapter_${profile} PRIVATE ../..)
    target_compile_definitions(uac_adapter_${profile} PRIVATE FM1_UAC_ADAPTER_TEST=1 FM1_USB_AUDIO_CAPTURE_TEST=1)
    if(profile STREQUAL "cdc_uvc")
        target_compile_definitions(uac_adapter_${profile} PRIVATE FM1_UVC_SCREEN=1)
    endif()
    target_compile_features(uac_adapter_${profile} PRIVATE c_std_11)
    if(MSVC)
        target_compile_options(uac_adapter_${profile} PRIVATE /W4 /WX)
    else()
        target_compile_options(uac_adapter_${profile} PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME adapter_${profile}_dma_iso_controls_shutdown COMMAND uac_adapter_${profile})
endforeach()
